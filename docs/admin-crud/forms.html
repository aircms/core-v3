Формы в админке

В админке форма валидирует данные и одновременно рисует интерфейс редактирования.

Свой класс формы

Рекомендуемый паттерн — наследоваться от Form и передавать элементы в parent::__construct(elements: [...]).

ProductForm
<?php

namespace App\Form\Admin;

use Air\Form\Form;
use Air\Form\Input;

class ProductForm extends Form
{
  public function __construct($model = null)
  {
    parent::__construct($model, elements: [
      Input::text('title'),
      Input::text('url'),
      Input::textarea('description', allowNull: true),
      Input::storage('image', allowNull: true),
      Input::checkbox('enabled'),
    ]);
  }
}
Использование
protected function getForm($model = null): ?Form
{
  return new \App\Form\Admin\ProductForm($model);
}

Изменить данные перед сохранением

getFormValues()
protected function getFormValues(Form $form): array
{
  $values = parent::getFormValues($form);

  if (empty($values['url'])) {
    $values['url'] = \Air\Slug::slugify($values['title']);
  }

  return $values;
}
getFormBlock()
protected function getFormBlock(ModelAbstract $model): ?string
{
  if (!$model->id) {
    return null;
  }

  return card(
    title: 'Statistics',
    content: vertical(content: [
      badge('Views: ' . (int)$model->views, LIGHT),
      badge('Orders: ' . (int)$model->orders, PRIMARY),
    ])
  );
}
