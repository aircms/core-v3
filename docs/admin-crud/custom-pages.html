Admin controller может содержать свои actions: import, preview, confirm, details, report, settings.

{
  $this->getView()->setVars([
    'title' => 'Import products',
    'form' => new \App\Form\Admin\ProductImportForm(),

  $this->getView()->setScript('product/import');

Кастомная форма не через manage()

ProductImportForm
<?php

namespace App\Form\Admin;

use Air\Form\Form;
use Air\Form\Input;

class ProductImportForm extends Form
{
  public function __construct()
  {
    parent::__construct(elements: [
      Input::storage('file'),
      Input::checkbox('dryRun'),
    ]);
  }
}
Обработка формы
public function import(): array|void
{
  $form = new \App\Form\Admin\ProductImportForm();

  if ($this->getRequest()->isPost()) {
    if (!$form->isValid($this->getRequest()->getPostAll())) {
      $this->getResponse()->setStatusCode(400);

      return [
        'errors' => $form->getErrorFields(),
      ];
    }

    $result = \App\Service\ProductImport::run($form->getCleanValues());

    return [
      'imported' => $result->imported,
      'skipped' => $result->skipped,
    ];
  }

  $this->getView()->setVars(['form' => $form]);
  $this->getView()->setScript('product/import');
}

Страница деталей

details action
public function details(\App\Model\Order $order): void
{
  $this->getView()->setVars([
    'order' => $order,
    'items' => $order->items,
    'history' => \App\Model\OrderHistory::fetchAll(['order' => $order]),
  ]);

  $this->getView()->setScript('order/details');
}
